CREATE TABLE PUBLISHER (
    CodigoPublisher INT PRIMARY KEY,
    Nombre VARCHAR2(255) NOT NULL,
    Region VARCHAR2(255),
    PrecioSuscripcion NUMBER(10,2),
    EnServicio VARCHAR2(50)
);
CREATE TABLE GENERO (
    IdGenero INT PRIMARY KEY,
    Nombre VARCHAR2(255) NOT NULL,
    Calificacion INT,
    CodigoPublisher INT,
    CONSTRAINT FK_GENERO_PUBLISHER FOREIGN KEY (CodigoPublisher) REFERENCES PUBLISHER(CodigoPublisher)
);
CREATE TABLE VIDEOJUEGO (
    CodigoSerie INT PRIMARY KEY,
    Nombre VARCHAR2(255) NOT NULL,
    Estado VARCHAR2(50),
    Entregado VARCHAR2(2), 
    Precio NUMBER(10,2),
    CodigoPublisher INT,
    CodigoVenta INT,
    CONSTRAINT FK_VIDEOJUEGOS_PUBLISHER FOREIGN KEY (CodigoPublisher) REFERENCES PUBLISHER(CodigoPublisher),
    CONSTRAINT FK_VIDEOJUEGOS_VENTA FOREIGN KEY (CodigoVenta) REFERENCES VENTA(CodigoVenta)
);
CREATE TABLE VENTA (
    CodigoVenta INT PRIMARY KEY,
    Fecha DATE,
    Cliente VARCHAR2(255),
    TotalVenta NUMBER(10,2)
);
